## Usage

```tsx
import {
  DndContext,
  DragOverlay,
  type DragEndEvent,
  type DragStartEvent,
} from "@dnd-kit/core"
import { arrayMove } from "@dnd-kit/sortable"

import { BoardCard } from "@/components/ui/board-card"
import { BoardColumn } from "@/components/ui/board-column"
```

```tsx
const [activeTask, setActiveTask] = useState<Task | null>(null)

function handleDragStart(event: DragStartEvent) {
  setActiveTask(tasks.find((t) => t.id === event.active.id) ?? null)
}

function handleDragEnd(event: DragEndEvent) {
  const { active, over } = event
  setActiveTask(null)
  if (!over || active.id === over.id) return
  const activeId = String(active.id)
  const overId = String(over.id)

  setTasks((prev) => {
    const activeIndex = prev.findIndex((t) => t.id === activeId)
    if (activeIndex === -1) return prev

    // Dropped on a column background (empty space) vs. on another task —
    // the latter is where in-column reordering comes from.
    const overIsColumn = STATUS_COLUMNS.includes(overId as TaskStatus)
    const overStatus = overIsColumn
      ? (overId as TaskStatus)
      : prev.find((t) => t.id === overId)?.status
    if (!overStatus) return prev

    const withStatus =
      prev[activeIndex].status === overStatus
        ? prev
        : prev.map((t) =>
            t.id === activeId ? { ...t, status: overStatus } : t
          )

    const from = withStatus.findIndex((t) => t.id === activeId)
    const to = overIsColumn
      ? withStatus.length - 1
      : withStatus.findIndex((t) => t.id === overId)
    return arrayMove(withStatus, from, to)
  })
}

;<DndContext id="tasks" onDragStart={handleDragStart} onDragEnd={handleDragEnd}>
  <div className="flex h-full gap-4">
    {STATUS_COLUMNS.map((status) => (
      <BoardColumn
        key={status}
        columnId={status}
        label={STATUS_LABEL[status]}
        items={tasks.filter((t) => t.status === status)}
        getItemId={(task) => task.id}
        renderItem={(task) => (
          <BoardCard itemId={task.id}>
            <span className="font-medium">{task.title}</span>
          </BoardCard>
        )}
      />
    ))}
  </div>
  <DragOverlay>
    {activeTask && (
      <BoardCard itemId={activeTask.id} overlay>
        <span className="font-medium">{activeTask.title}</span>
      </BoardCard>
    )}
  </DragOverlay>
</DndContext>
```

Wrap every column for a board in a single `DndContext` — one context per
board, not per column. Pass a stable `id` (dnd-kit falls back to a
module-level counter for its `DndDescribedBy-N` accessibility id otherwise,
which drifts between server and client renders — a second board on the same
page, or React Strict Mode's dev-only double-invoke, is enough to trigger a
hydration mismatch). Each `BoardColumn` registers itself as a drop target
(dnd-kit's `useDroppable({ id: columnId })`) and, internally, wraps its own
items in a `SortableContext` keyed by item id — that's what makes a card
droppable onto another card, not just onto the column. The header shows
`label` plus an item-count badge, the body lists `items` via `renderItem`,
and drops fall back to `emptyLabel` when the column is empty.

`DndContext`'s `onDragEnd` is where you move (and reorder) the item in your
own state — it fires with the dragged item's id on `event.active.id`
(written by [`BoardCard`](/docs/components/board-card)) and whatever it was
dropped on via `event.over.id`, which is either a column id (dropped on empty
column space — append to the end) or another item's id (dropped on a card —
that's the in-column/cross-column reorder case). Tell the two apart by
checking `event.over.id` against your column list, then use `arrayMove` to
place the item at the right position — `BoardColumn` doesn't track order
itself, so the position of each item in your own array is what's rendered.

Add a `DragOverlay` (and its `onDragStart` bookkeeping) so the dragged card
renders in a portal above everything instead of clipping against a
column's `overflow-y-auto` — pass `overlay` to the `BoardCard` rendered
inside it so it skips its own drag listeners (see
[`BoardCard`](/docs/components/board-card#api-reference)).

### Reorderable columns

Pass `sortable` to let a column be dragged by a grip handle in its header
(instead of the whole header) to reorder the board itself. It needs
`@dnd-kit/sortable`'s `SortableContext` wrapping the columns — same
`DndContext`, no second one:

```tsx
import {
  SortableContext,
  arrayMove,
  horizontalListSortingStrategy,
} from "@dnd-kit/sortable"
```

```tsx
function handleDragEnd(event: DragEndEvent) {
  const { active, over } = event
  setActiveTask(null)
  if (!over || active.id === over.id) return
  const activeId = String(active.id)
  const overId = String(over.id)

  // Column ids and item ids share this DndContext — tell them apart by
  // which list `active.id` belongs to before deciding what moved.
  if (columnOrder.includes(activeId as TaskStatus)) {
    setColumnOrder((prev) =>
      arrayMove(
        prev,
        prev.indexOf(activeId as TaskStatus),
        prev.indexOf(overId as TaskStatus)
      )
    )
    return
  }

  // ...same item move/reorder logic as above, unchanged
}

;<DndContext onDragStart={handleDragStart} onDragEnd={handleDragEnd}>
  <SortableContext items={columnOrder} strategy={horizontalListSortingStrategy}>
    <div className="flex h-full gap-4">
      {columnOrder.map((status) => (
        <BoardColumn
          key={status}
          columnId={status}
          sortable /* ...rest unchanged */
        />
      ))}
    </div>
  </SortableContext>
  <DragOverlay>{/* ...unchanged */}</DragOverlay>
</DndContext>
```

`BoardColumn` doesn't own the column order itself — same rule as items:
`onDragEnd` decides what moved and updates your own state, here a
`columnOrder` array reordered with `arrayMove`.

## API Reference

| Prop          | Type                               | Default               | Description                                                                                              |
| ------------- | ---------------------------------- | --------------------- | -------------------------------------------------------------------------------------------------------- |
| `columnId`    | `string`                           | -                     | This column's droppable id (`useDroppable({ id: columnId })`) — read off `event.over.id` in `onDragEnd`. |
| `label`       | `string`                           | -                     | Column header text.                                                                                      |
| `items`       | `TItem[]`                          | -                     | Items to render in this column.                                                                          |
| `getItemId`   | `(item: TItem) => string`          | -                     | Extracts a stable key from each item.                                                                    |
| `renderItem`  | `(item: TItem) => React.ReactNode` | -                     | Renders each item — typically a `BoardCard`.                                                             |
| `emptyLabel`  | `string`                           | `"Drop an item here"` | Text shown when `items` is empty.                                                                        |
| `headerExtra` | `React.ReactNode`                  | -                     | Rendered in the header after the count badge (e.g. a running total).                                     |
| `className`   | `string`                           | -                     | Additional CSS classes on the column's outer element.                                                    |
| `sortable`    | `boolean`                          | -                     | Makes the column draggable via a header grip handle; needs a `SortableContext` wrapping the columns.     |
